cmake_minimum_required(VERSION 3.20)
project(app C ASM)
set(CMAKE_C_STANDARD 11)
set(STM32_CHIP STM32F407xx)
add_definitions(-D${STM32_CHIP})
# Coverage support (ENABLE_COVERAGE=ON → -fprofile-arcs -ftest-coverage)
if(ENABLE_COVERAGE)
  set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} --coverage -O0 -g")
  set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} --coverage")
  message(STATUS "ENABLE_COVERAGE: instrumentation ON")
endif()
add_executable(app.elf src/main.c)
