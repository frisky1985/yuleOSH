cmake_minimum_required(VERSION 3.16)
project(app C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -mcpu=cortex-m4 -mthumb -Wall -Wextra -Werror")
add_compile_definitions(SAFETY_ASIL_B)
# Coverage support (ENABLE_COVERAGE=ON → -fprofile-arcs -ftest-coverage)
if(ENABLE_COVERAGE)
  set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} --coverage -O0 -g")
  set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} --coverage")
  message(STATUS "ENABLE_COVERAGE: instrumentation ON")
endif()
add_executable(app.elf src/main.c)
